== Direct writes that never go through the guard ==
-- 1. Into a program folder --
$ bash zone-run.sh sh -c 'echo x > content/direct.md'
sh: content/direct.md: Operation not permitted
exit code: 1

-- 2. Appending to an existing program file --
$ bash zone-run.sh sh -c 'echo more >> content/quest.md'
sh: content/quest.md: Operation not permitted
exit code: 1

-- 3. A new file at the top of the repository --
$ bash zone-run.sh sh -c 'touch direct.md'
touch: direct.md: Operation not permitted
exit code: 1

-- 4. Into the Generated zone --
$ bash zone-run.sh sh -c 'echo x > generated/direct.md'
sh: generated/direct.md: Operation not permitted
exit code: 1

-- 5. A new folder in no zone --
$ bash zone-run.sh sh -c 'mkdir scratch2'
mkdir: scratch2: Operation not permitted
exit code: 1

-- 6. Using .. to leave the repository --
$ bash zone-run.sh sh -c 'echo x > participant/../../outside/direct.md'

exit code: 1

-- 7. Through the link that leads outside --
$ bash zone-run.sh sh -c 'echo x > participant/outside-link/direct.md'

exit code: 1

-- 8. Through the link into content/ --
$ bash zone-run.sh sh -c 'echo x > participant/content-link/direct.md'

exit code: 1

-- 9. Copying into a program folder --
$ bash zone-run.sh sh -c 'cp README.md content/copy.md'

exit code: 1

-- 10. Deleting a program file --
$ bash zone-run.sh sh -c 'node -e "require(\"fs\").unlinkSync(\"content/quest.md\")" 2>&1 | grep -o "EPERM: operation not permitted"; exit 1'

-- 12. Into local-data/ (allowed) --
