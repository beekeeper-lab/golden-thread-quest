@test
def raw_transcript_git_would_commit_is_refused(ws):
    """AC 10 and the safety constraint: inside the repository, a raw transcript is accepted only
    when git ignores it (SKILL.md: refused if "inside the repository and not ignored by git").
    Covers: no repository git can read, no ignore rule, ignored but already tracked, and ignored."""
    module = ws.module
    module.REPO = ws.root
    module.PARTICIPANT = ws.root / "participant"
    raw, meeting = ws.raw(BASIC)

    def git(*args):
        subprocess.run(["git", "-C", str(ws.root), *args], check=True, capture_output=True)

    def refused(case):
        try:
            module.ingest(raw, meeting, ws.out_dir / "sample.md", output_dir=ws.out_dir)
        except module.InvalidInput as error:
            expect("does not ignore" in str(error), f"{case}: refused for the wrong reason: {error}")
        else:
            raise AssertionError(f"{case}: the raw transcript was accepted, but git does not "
                                 f"ignore it; wrote {ws.files()}")
        expect(ws.files() == [], f"{case}: refused, but left files behind: {ws.files()}")

    (ws.root / ".git").write_text("gitdir: no-such-folder\n", encoding="utf-8")
    refused("a repository git cannot read (git exits with an error)")
    (ws.root / ".git").unlink()
    git("init", "-q")
    refused("a repository with no ignore rule")
    git("add", "raw/meeting.vtt")
    (ws.root / ".gitignore").write_text("/raw/\n", encoding="utf-8")
    refused("matched by .gitignore but already tracked, so git still commits its changes")
    git("rm", "-q", "--cached", "raw/meeting.vtt")
    module.ingest(raw, meeting, ws.out_dir / "sample.md", output_dir=ws.out_dir)
    expect(ws.files() == ["sample.md", "sample.provenance.yaml"],
           f"an ignored, untracked raw transcript was refused or wrote the wrong files: {ws.files()}")

class DiskFull(OSError):
    pass


class Interrupted(KeyboardInterrupt):
    pass


class HalfWriter:
    def __init__(self, stream):
        self.stream = stream
    def __enter__(self):
        return self
    def __exit__(self, *exc):
        self.stream.close()
        return False
    def write(self, text):
        self.stream.write(text[: len(text) // 2])

**T55. A real hang never reaches the test's check; the suite blocks with no message**
- Test: hostile_text.
- Agreed: yes. Broader: under load, all three planted slowness bugs show up as long waits (120 to 188 s) or as silent kills.

```python
class Stalled(Exception):
    pass


@contextlib.contextmanager
def within_seconds(limit, label):
    """Fail, with a message, when the block runs past the limit; POSIX main thread only."""
    import signal
    if not hasattr(signal, "setitimer"):
        yield
        return
    def stalled(signum, frame):
        raise Stalled()
    previous = signal.signal(signal.SIGALRM, stalled)
    outer, _ = signal.setitimer(signal.ITIMER_REAL, limit)
    started = time.monotonic()
    try:
        yield
    except Stalled:
        raise AssertionError(f"{label} did not finish within {limit} seconds; it was stopped") from None
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0)
        signal.signal(signal.SIGALRM, previous)
        if outer:
            signal.setitimer(signal.ITIMER_REAL, max(outer - (time.monotonic() - started), 0.001))
```
